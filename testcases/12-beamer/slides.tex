\section{Outline behaviour}
\begin{frame}[fragile]{Frames are not sections}
  \begin{itemize}
    \item The outline sidebar lists \verb|\section| but probably not frames.
    \item Record whether you would want frame titles in the outline.
  \end{itemize}
\end{frame}

\begin{frame}{Overlays}
  \begin{enumerate}
    \item<1-> First appears on slide 1
    \item<2-> Second appears on slide 2
    \item<3-> Third appears on slide 3
  \end{enumerate}
  One frame -> three PDF pages. Page counter in the status bar must say 3 / N
  pages for a single frame; SyncTeX from this line should go to the first of them.
\end{frame}

\section{Math}

  \begin{equation}
    \nabla \cdot \mathbf{E} = \frac{\rho}{\varepsilon_0}
  \end{equation}

\end{frame}
